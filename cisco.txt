                                      !Part1 - Configuration Commands:
                     !---------------ROUTER 1----------------------------
En
Config t
hostname R1
int s0/3/0 
ip add 60.0.0.1 255.255.255.252
no shut
exit
int gig0/0
                                      ! adjusted for /24
ip add 10.0.0.1 255.255.255.0
no shut 
exit
int gig0/1
ip add 20.0.0.1 255.255.255.0
no shut
exit

! R1
ip dhcp excluded-address 10.0.0.1
ip dhcp excluded-address 20.0.0.1
ip route 70.0.0.0 255.255.255.252 60.0.0.2

service dhcp 
ip dhcp pool pool1
                                      ! adjusted for /24
network 10.0.0.0 255.255.255.0
default-router 10.0.0.1
exit
ip dhcp pool pool2
network 20.0.0.0 255.255.255.0
default-router 20.0.0.1
exit

! 04.1 - R1 RIP CONFIGURATION
router rip
version 2
no auto-summary
network 10.0.0.0
network 20.0.0.0
network 60.0.0.0
exit

! Router 1
router bgp 65001
neighbor 70.0.0.2 remote-as 65002
neighbor 70.0.0.2 ebgp-multihop 2
network 10.0.0.0 mask 255.255.255.0
network 20.0.0.0 mask 255.255.255.0
exit

! R1 SSH CONFIGURATION
ip domain-name cisco.com
username admin privilege 15 secret cisco
crypto key generate rsa
! Choose 2048 bits when prompted
ip ssh version 2
line vty 0 4
login local
transport input ssh
exit
line vty 5 15
login local
transport input ssh
exit
end
write memory
