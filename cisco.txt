! R1 LAN 1 = 10.0.0.0/24 - Gateway 10.0.0.1   ! 01: IP ADDRESSING PLAN
! R1 LAN 2 = 20.0.0.0/24 - Gateway 20.0.0.1

! R2 LAN 1 = 30.0.0.0/24 - Gateway 30.0.0.1
! R2 LAN 2 = 40.0.0.0/24 - Gateway 40.0.0.1

! R3 LAN 1 = 50.0.0.0/24 - Gateway 50.0.0.1

! R1 TO R2 = 60.0.0.0/30
! R2 TO R3 = 70.0.0.0/30
              
En                                            ! 02: R1 Configuration
Config t
hostname R1
int s0/3/0 
ip add 60.0.0.1 255.255.255.252
no shut
exit
int gig0/0
                                    
ip add 10.0.0.1 255.255.255.0
no shut 
exit
int gig0/1
ip add 20.0.0.1 255.255.255.0
no shut
exit

ip dhcp excluded-address 10.0.0.1           ! 02.3: DHCP Configuration
ip dhcp excluded-address 20.0.0.1
ip route 70.0.0.0 255.255.255.252 60.0.0.2

service dhcp 
ip dhcp pool pool1
                                     
network 10.0.0.0 255.255.255.0
default-router 10.0.0.1
exit
ip dhcp pool pool2
network 20.0.0.0 255.255.255.0
default-router 20.0.0.1
exit

router rip                                ! 02.4: R1 - RIP V2
version 2
no auto-summary
network 10.0.0.0
network 20.0.0.0
network 60.0.0.0
exit

router bgp 65001                          ! 02.5: R1 BGP Configuration
neighbor 70.0.0.2 remote-as 65002
network 10.0.0.0 mask 255.255.255.0
network 20.0.0.0 mask 255.255.255.0
exit

ip domain-name cisco.com                  ! 02.6: R1 SSH CONFIGURATION
username admin privilege 15 secret cisco
crypto key generate rsa                  ! Choose 2048 bits when prompted, paste upto here
                                          
ip ssh version 2
line vty 0 4
login local
transport input ssh
exit
line vty 5 15
login local
transport input ssh
exit
end
write memory
                      
En                                        !03: ROUTER 2
Config t
hostname R2
int s0/3/0 
ip add 60.0.0.2 255.255.255.252
no shut
exit
int gig0/0
                             
ip add 30.0.0.1 255.255.255.0      
no shut 
exit
int gig0/1
ip add 40.0.0.1 255.255.255.0
no shut
exit
int s0/3/1
ip add 70.0.0.1 255.255.255.252
no shut
exit

ip dhcp excluded-address 30.0.0.1
ip dhcp excluded-address 40.0.0.1

service dhcp 
ip dhcp pool pool1
network 30.0.0.0 255.255.255.0
default-router 30.0.0.1
exit
ip dhcp pool pool2
network 40.0.0.0 255.255.255.0
default-router 40.0.0.1
exit
                 
router rip                                    ! 03.3: R2 RIP CONFIGURATION
version 2
no auto-summary
network 30.0.0.0
network 40.0.0.0
network 60.0.0.0
exit

router ospf 1                                  ! 03.4: R2 OSPF CONFIGURATION
network 30.0.0.0 0.0.0.255 area 0
network 40.0.0.0 0.0.0.255 area 0
network 70.0.0.0 0.0.0.3 area 0
exit

ip domain-name cisco.com                      ! 03.5: R2 SSH CONFIGURATION
username admin privilege 15 secret cisco
crypto key generate rsa                      ! Choose 2048 bits when prompted, paste upto here

ip ssh version 2
line vty 0 4
login local
transport input ssh
exit
line vty 5 15
login local
transport input ssh
exit
end
write memory
                                              ! 04: ROUTER 3
En

ip add 70.0.0.2 255.255.255.252
no shut
exit
int gig0/0

no shut
exit

exit

transport input ssh
exit
line vty 5 15
login local
transport input ssh
exit
end
